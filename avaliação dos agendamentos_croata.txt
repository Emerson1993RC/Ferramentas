<script>
  const FAIXAS = [
    { label: 'Ótimo',      min: 50, max: 70,  cor: '#0f6e56', bg: '#e1f5ee', classe: 'faixa-otimo',      intervalo: '> 50 e ≤ 70' },
    { label: 'Bom',        min: 30, max: 50,  cor: '#185fa5', bg: '#e6f1fb', classe: 'faixa-bom',        intervalo: '> 30 e ≤ 50' },
    { label: 'Suficiente', min: 10, max: 30,  cor: '#633806', bg: '#faeeda', classe: 'faixa-suficiente', intervalo: '> 10 e ≤ 30' },
    { label: 'Regular',    min: -1, max: 10,  cor: '#791f1f', bg: '#fcebeb', classe: 'faixa-regular',    intervalo: '≤ 10 ou > 70' },
  ];

  /* Mensagens dinâmicas por faixa */
  const ALERTAS = {
    'Ótimo': {
      cls: 'alert-otimo',
      titulo: 'Ótimo desempenho — mantenha o ritmo!',
      msg: 'A proporção de consultas programadas está no intervalo ideal (entre 50% e 70%). Continue equilibrando os agendamentos programados com o atendimento à demanda espontânea. Não reduza as consultas no dia — o equilíbrio é o que garante essa classificação.'
    },
    'Bom': {
      cls: 'alert-bom',
      titulo: 'Bom desempenho — aumentar agendamentos programados.',
      msg: 'A proporção de consultas programadas está boa, mas ainda abaixo do intervalo ótimo (50–70%). Priorize o agendamento de consultas programadas e de cuidado continuado para pacientes com acompanhamento regular: hipertensos, diabéticos, gestantes e crianças.'
    },
    'Suficiente': {
      cls: 'alert-suficiente',
      titulo: 'Atenção — aumentar agendamentos programados com urgência.',
      msg: 'A proporção de consultas programadas está baixa. É necessário ampliar o agendamento de consultas programadas. Converse com a equipe sobre os pacientes que precisam de acompanhamento contínuo e que ainda não têm consulta marcada.'
    },
    'Regular': {
      cls: 'alert-regular',
      titulo: 'Resultado irregular — ação imediata necessária.',
      msg: 'O resultado está fora dos parâmetros aceitáveis (≤ 10% ou > 70%). Se muito baixo, a unidade pode estar centrada apenas em urgências. Se muito alto, pode estar fechada à demanda espontânea. Avalie com a coordenação qual ajuste é necessário.'
    }
  };

  const PROGRAMADA = ['Consulta Agendada', 'Consulta Agendada Programada / Cuidado Continuado'];
  const ESPONTANEA = ['Consulta no Dia', 'Atendimento de Urgência'];

  function classificar(v) {
    if (v > 70) return FAIXAS[3];
    return FAIXAS.find(f => v > f.min && v <= f.max) || FAIXAS[3];
  }

  function fmt(n) { return n.toFixed(2).replace('.', ','); }

  function parseCSV(text) {
    /* Normaliza quebras de linha e remove BOM */
    text = text.replace(/^\uFEFF/, '').replace(/\r\n/g, '\n').replace(/\r/g, '\n');
    const linhas = text.split('\n');

    let periodo = '', geradoEm = '', tabelaIdx = -1;
    const dados = {};

    for (let i = 0; i < linhas.length; i++) {
      const l = linhas[i];
      if (l.startsWith('Período:')) {
        const m = l.match(/(\d{2}\/\d{2}\/\d{4} a \d{2}\/\d{2}\/\d{4})/);
        if (m) periodo = m[1];
      }
      if (l.startsWith('Gerado em')) geradoEm = l.replace('Gerado em ', '').trim();
      if (l.includes('Tipo de Atendimento') && l.includes('Totais')) tabelaIdx = i;
    }

    if (tabelaIdx >= 0) {
      /* Cabeçalho: ;Tipo de Atendimento;Urgência;Agendada;AgendadaProg;ConsultaDia;Totais; */
      const cab = linhas[tabelaIdx].split(';').map(c => c.trim());

      for (let i = tabelaIdx + 1; i < linhas.length; i++) {
        const raw = linhas[i];
        if (!raw.trim()) continue;

        const cols = raw.split(';').map(c => c.replace(',', '.').trim());

        /* Linha de dados: "Atendimento Individual;;val;val;val;val;val;" */
        const isData = cols[0] === 'Atendimento Individual';
        /* Linha de totais: "Totais;;val;val;val;val;val;" */
        const isTotals = cols[0] === 'Totais';

        if (isData || isTotals) {
          /* cab[0]="" cab[1]="Tipo de Atendimento" cab[2]="Urgência" ... cab[N]="Totais"
             cols[0]="Atendimento Individual" cols[1]="" cols[2]=val ... */
          for (let j = 2; j < cab.length; j++) {
            const nome = cab[j];
            const val = parseFloat(cols[j]);
            if (nome && !isNaN(val)) {
              dados[nome] = val;
            }
          }
          if (isData) continue; /* continua para pegar linha Totais também */
          if (isTotals) break;
        }
      }
    }
    return { periodo, geradoEm, dados };
  }

  function renderizar({ periodo, geradoEm, dados }) {
    if (periodo) document.getElementById('subtitulo').textContent =
      'Município de Varjota · Ceará · Período: ' + periodo;
    if (geradoEm) {
      const dataHora = geradoEm.match(/(\d{2}\/\d{2}\/\d{4} às \d{2}:\d{2})/);
      document.getElementById('rodape').textContent =
        dataHora ? 'Gerado em ' + dataHora[1] : 'Gerado em ' + geradoEm;
    }

    const progTotal = PROGRAMADA.reduce((s, k) => s + (dados[k] || 0), 0);
    const espoTotal = ESPONTANEA.reduce((s, k) => s + (dados[k] || 0), 0);
    const faixa = classificar(progTotal);

    /* Alerta dinâmico */
    const alInfo = ALERTAS[faixa.label];
    const alEl = document.getElementById('alerta');
    alEl.className = 'alert ' + alInfo.cls;
    alEl.innerHTML = `<strong>${alInfo.titulo}</strong>${alInfo.msg}`;

    /* Cards */
    document.getElementById('val-c1').textContent = fmt(progTotal) + '%';
    document.getElementById('sub-c1').textContent = 'Demanda programada sobre o total';
    document.getElementById('card-c1').className = 'card ' + faixa.classe;
    document.getElementById('val-faixa').textContent = faixa.label;
    document.getElementById('card-faixa').className = 'card ' + faixa.classe;
    document.getElementById('val-espontanea').textContent = fmt(espoTotal) + '%';
    document.getElementById('val-agendada').textContent =
      fmt(dados['Consulta Agendada'] || 0) + '%';

    /* Tabela — ordem fixa para evitar inversão de colunas */
    const ordem = [
      'Consulta Agendada',
      'Consulta Agendada Programada / Cuidado Continuado',
      'Consulta no Dia',
      'Atendimento de Urgência',
    ];
    const maxVal = Math.max(...ordem.map(k => dados[k] || 0));
    let html = '';
    for (const tipo of ordem) {
      const val = dados[tipo] || 0;
      const pct = maxVal > 0 ? (val / maxVal) * 100 : 0;
      const isProg = PROGRAMADA.includes(tipo);
      const label = tipo === 'Consulta Agendada Programada / Cuidado Continuado'
        ? 'Cons. Agendada Programada / CC' : tipo;
      html += `<tr>
        <td>${label}</td>
        <td style="text-align:right">${fmt(val)}%</td>
        <td><div class="bar-wrap"><div class="bar-fill ${isProg ? 'bar-prog' : 'bar-espo'}"
          style="width:${pct.toFixed(0)}%"></div></div></td>
        <td style="text-align:right; font-size:11px; color:#6b6b65">
          ${isProg ? 'Programada' : 'Espontânea'}</td>
      </tr>`;
    }
    const tot = dados['Totais'] || 0;
    html += `<tr class="totals">
      <td>Totais</td>
      <td style="text-align:right">${fmt(tot)}%</td>
      <td></td><td></td>
    </tr>`;
    document.getElementById('tabela-body').innerHTML = html;

    /* Legenda faixas */
    document.getElementById('legenda').innerHTML = FAIXAS.map(f => {
      const atual = f.label === faixa.label;
      return `<div class="leg-item" style="${atual
        ? 'border-color:'+f.cor+'; background:'+f.bg+'; border-width:1.5px' : ''}">
        <div class="leg-dot" style="background:${f.cor}"></div>
        <div>
          <div style="font-size:12px; color:${atual ? f.cor : '#1a1a18'};
            font-weight:${atual ? 600 : 400}">${f.label}${atual ? ' ← atual' : ''}</div>
          <div style="font-size:11px; color:#888780">${f.intervalo}</div>
        </div>
      </div>`;
    }).join('');

    document.getElementById('conteudo').style.display = 'block';
  }

  function processarArquivo(file) {
    if (!file || !file.name.endsWith('.csv')) return;
    const area = document.getElementById('upload-area');
    area.classList.add('carregado');
    document.getElementById('upload-label').textContent = 'Relatório carregado com sucesso';
    document.getElementById('upload-sub').textContent = 'Clique aqui para trocar por outro arquivo CSV';
    document.getElementById('upload-nome').textContent = file.name;
    document.getElementById('upload-svg').innerHTML = '<polyline points="20 6 9 17 4 12"/>';

    const reader = new FileReader();
    reader.onload = e => renderizar(parseCSV(e.target.result));
    reader.readAsText(file, 'UTF-8');
  }

  document.getElementById('file-input').addEventListener('change', e => {
    processarArquivo(e.target.files[0]);
  });

  const area = document.getElementById('upload-area');
  area.addEventListener('dragover', e => { e.preventDefault(); area.classList.add('drag'); });
  area.addEventListener('dragleave', () => area.classList.remove('drag'));
  area.addEventListener('drop', e => {
    e.preventDefault(); area.classList.remove('drag');
    processarArquivo(e.dataTransfer.files[0]);
  });
</script>
